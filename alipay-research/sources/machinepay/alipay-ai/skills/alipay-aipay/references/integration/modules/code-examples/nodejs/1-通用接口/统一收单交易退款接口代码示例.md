# 统一收单交易退款接口 - Node.js 示例

## 请求示例

```javascript
const { AlipaySdk } = require("alipay-sdk");

const alipaySdk = new AlipaySdk({
  appId: "<-- 请填写您的AppId，例如：20190xxxxxx -->",
  privateKey: "<-- 请填写您的应用私钥，例如：MIIE...（脱敏示例） -->", // Node.js 使用 PKCS#1 格式私钥
  alipayPublicKey: "<-- 请填写您的支付宝公钥，例如：MIIB...（脱敏示例） -->",
  gateway: "https://openapi.alipay.com/gateway.do",
  signType: "RSA2",
  // 第三方代调用模式下请设置 appAuthToken
  // appAuthToken: "<-- 请填写应用授权令牌 -->",
});

const result = await alipaySdk.exec("alipay.trade.refund", {
  bizContent: {
    out_trade_no: "MERCHANT_ORDER_NO_PLACEHOLDER",  // 商户订单号，与 trade_no 二选一
    refund_amount: "200.12",             // 使用十进制定点金额；调用前校验累计退款不超过原订单实付金额
    refund_reason: "正常退款",            // 退款原因，选填
    out_request_no: "OUT_REQUEST_NO_PLACEHOLDER", // 新退款使用新编号；同一退款失败重试必须复用原编号和原参数
  },
});

if (result.code === "10000") {
  if (result.fund_change === "Y") {
    console.log("退款资金已发生变化，退款金额:", result.refund_fee);
  } else {
    console.log("退款结果待确认：使用同一 out_request_no 调用 alipay.trade.fastpay.refund.query");
  }
} else {
  console.log("调用失败");
}
```

## 响应示例
### 正常示例
```json
{
    "code": "10000",
    "msg": "Success",
    "trade_no": "ALIPAY_TRADE_NO_PLACEHOLDER",
    "out_trade_no": "MERCHANT_ORDER_NO_PLACEHOLDER",
    "buyer_logon_id": "159****5620",
    "fund_change": "Y",
    "refund_fee": "88.88"
}
```
### 异常示例
```json
{
    "code": "20000",
    "msg": "Service Currently Unavailable",
    "sub_code": "isp.unknow-error",
    "sub_msg": "系统繁忙"
}
```
